Class ModelStore:
  - __init__: self._models = {} (empty dict), load_all_models() called in __init__

  Method load_all_models():
    - Log "Loading all ML models..." to system_logger
    - For each model_name in VALID_MODEL_NAMES:
      - Construct path: Path(MODELS_DIR) / f"{model_name}.pkl"
      - Try: load with joblib.load(path)
      - Store in self._models[model_name]
      - Log success: f"Loaded {model_name} from {path}" → system_logger
      - Except FileNotFoundError: log error → error_logger, set self._models[model_name] = None
      - Except Exception as e: log error with traceback → error_logger, set None
    - Log "Model loading complete. {loaded_count}/{total} models loaded." → system_logger

  Method get_model(model_name: str):
    - If model_name not in VALID_MODEL_NAMES:
      raise ValueError(f"Unknown model: {model_name}. Valid: {VALID_MODEL_NAMES}")
    - If self._models.get(model_name) is None:
      raise RuntimeError(f"Model '{model_name}' failed to load at startup.")
    - Return self._models[model_name]

  Method is_loaded(model_name: str) → bool:
    Return self._models.get(model_name) is not None

  Method get_loaded_models() → list[str]:
    Return [k for k, v in self._models.items() if v is not None]

  Method get_metadata(model_name: str) → dict:
    Return MODEL_METADATA.get(model_name, {})

Create module-level singleton: model_store = ModelStore()
Export: model_store, MODEL_METADATA, VALID_MODEL_NAMES

class ModuleAFeatures(BaseModel):
    Cement: float
    Blast_Furnace_Slag: float = Field(alias="Blast Furnace Slag")
    Fly_Ash: float = Field(alias="Fly Ash")
    Water: float
    Superplasticizer: float
    Coarse_Aggregate: float = Field(alias="Coarse Aggregate")
    Fine_Aggregate: float = Field(alias="Fine Aggregate")
    Age: float
    model_config = ConfigDict(populate_by_name=True)

    Validators (use @field_validator):
      Cement: must be between 100 and 600
      Water: must be between 100 and 250
      Age: must be between 1 and 365
      All others: must be >= 0

━━━━━━━━━━━
ENDPOINT: POST /predict/{model_name}
━━━━━━━━━━━

Path param: model_name: str
Body: PredictRequest (raw dict of features)
Returns: PredictResponse

Full implementation logic:

Step 1 — Validate model_name:
  If model_name not in VALID_MODEL_NAMES:
    raise HTTPException(status_code=404,
      detail=f"Model '{model_name}' not found. Valid models: {VALID_MODEL_NAMES}")

Step 2 — Get model:
  Try: model = model_store.get_model(model_name)
  Except RuntimeError as e:
    error_logger.error(f"Model retrieval failed: {e}")
    raise HTTPException(status_code=503, detail=str(e))

Step 3 — Parse and validate features based on model_name:

  If model_name == "module_a_rf":
    Try: parsed = ModuleAFeatures(**request.features)
    input_array = np.array([[
      parsed.Cement,
      getattr(parsed, 'Blast_Furnace_Slag'),
      getattr(parsed, 'Fly_Ash'),
      parsed.Water,
      parsed.Superplasticizer,
      getattr(parsed, 'Coarse_Aggregate'),
      getattr(parsed, 'Fine_Aggregate'),
      parsed.Age
    ]])
    IMPORTANT: respect the exact column order the model was trained on:
    [Cement, Blast Furnace Slag, Fly Ash, Water, Superplasticizer,
     Coarse Aggregate, Fine Aggregate, Age]

  If model_name == "module_b_xgb":
    Try: parsed = ModuleBFeatures(**request.features)
    input_array = np.array([[
      parsed.C, parsed.Mn, parsed.Si, parsed.Cr, parsed.Ni,
      parsed.Mo, parsed.Cu, parsed.V, parsed.Al, parsed.Ti,
      parsed.N, parsed.B, parsed.Reduction_Ratio
    ]])

  If model_name in ["module_c_reg", "module_c_clf"]:
    Try: parsed = ModuleCFeatures(**request.features)
    input_array = np.array([[
      parsed.Su, parsed.Sy, parsed.Hardness,
      parsed.Elongation, parsed.Reduction_of_Area
    ]])

  Except (ValidationError, ValueError) as e:
    error_logger.error(f"Input validation failed for {model_name}: {e}")
    raise HTTPException(status_code=422, detail=str(e))

Step 4 — Run prediction:
  Try:
    raw_prediction = model.predict(input_array)

    If model_name == "module_a_rf":
      prediction_value = float(round(raw_prediction[0], 4))
      confidence = None
      task = "regression"
      target = "Compressive Strength (MPa)"

    If model_name == "module_b_xgb":
      prediction_value = float(round(raw_prediction[0], 4))
      confidence = None
      task = "regression"
      target = "UTS (MPa)"

    If model_name == "module_c_reg":
      # raw_prediction shape: (1, N_targets)
      # Keys are the regression targets in training order: Su, Sy
      pred_array = raw_prediction[0]
      prediction_value = {
        "Su": float(round(pred_array[0], 4)),
        "Sy": float(round(pred_array[1], 4))
      }
      confidence = None
      task = "multi-output regression"
      target = "Su (MPa), Sy (MPa)"

    If model_name == "module_c_clf":
      prediction_value = str(raw_prediction[0])
      # Attempt to get probability confidence
      If hasattr(model, "predict_proba"):
        proba = model.predict_proba(input_array)[0]
        confidence = float(round(max(proba), 4))
      Else: confidence = None
      task = "classification"
      target = "Use (Not Used / Used)"
      # Map numeric label to string if needed:
      label_map = {0: "Not Used", 1: "Used"}
      If raw_prediction[0] in label_map:
        prediction_value = label_map[raw_prediction[0]]

  Except Exception as e:
    error_logger.error(f"Prediction failed for {model_name}: {e}", exc_info=True)
    raise HTTPException(status_code=500,
      detail=f"Prediction failed: {str(e)}")

Step 5 — Log prediction:
  prediction_logger.info(
    f"model={model_name} | input={request.features} | prediction={prediction_value}"
  )

Step 6 — Return PredictResponse:
  Return PredictResponse(
    prediction=prediction_value,
    model_used=model_name,
    confidence=confidence,
    metrics=model_store.get_metadata(model_name),
    task=task,
    target=target
  )
